def norm_lines(text):
    lines = [l.rstrip() for l in text.replace('\r\n', '\n').replace('\r', '\n').split('\n')]
    while lines and not lines[-1]:
        lines.pop()
    return lines


def h(s):
    return hashlib.sha256(s.encode('utf-8')).hexdigest()


def hnorm(text):
    return h('\n'.join(norm_lines(text)))


def run(args, cwd):
    r = subprocess.run([sys.executable] + args, capture_output=True, text=True, cwd=cwd)
    return r.returncode, r.stdout, r.stderr


def check_files():
    bad = []
    for rel, m in sorted(MANIFEST.items()):
        path = os.path.join(SKILL, *rel.split('/'))
        if not os.path.exists(path):
            bad.append(f'MISSING  {rel}')
            continue
        with open(path, encoding='utf-8', errors='replace') as f:
            lines = norm_lines(f.read())
        if h('\n'.join(lines)) == m['sha']:
            continue
        where = ''
        for i, ch in enumerate(m['chunks']):
            if h('\n'.join(lines[i * CHUNK:(i + 1) * CHUNK]))[:10] != ch:
                where = f'first diff in lines {i * CHUNK + 1}-{(i + 1) * CHUNK}'
                break
        if not where:
            where = 'extra lines at the end'
        bad.append(f'CHANGED  {rel}: {where} (has {len(lines)} lines, expected {m["lines"]})')
    return bad


def behaviour():
    """Return {name: hash} for everything the tools produce on the built-in inputs."""
    out = {}
    tmp = tempfile.mkdtemp(prefix='bp_selftest_')
    try:
        for ex in ('minimal_process.py', 'minimal_orchestration.py'):
            code, so, se = run([os.path.join(SKILL, 'examples', ex)], tmp)
            out[f'run {ex}'] = 'ok' if code == 0 else f'exit {code}: {se.strip()[-120:]}'
        xml_dir = os.path.join(tmp, 'out')
        for fn in sorted(os.listdir(xml_dir)) if os.path.isdir(xml_dir) else []:
            with open(os.path.join(xml_dir, fn), encoding='utf-8') as f:
                out[f'xml {fn}'] = hnorm(f.read())
        code, so, se = run([os.path.join(SKILL, 'lib', 'validate.py'), 'out', '--flow', '--estate',
                            '--strict'], tmp)
        out['validate report'] = hnorm(so)
        out['validate exit'] = str(code)
        with open(os.path.join(tmp, 'fixture.ini'), 'w', encoding='utf-8', newline='\n') as f:
            f.write(FIXTURE_INI)
        code, so, se = run([os.path.join(SKILL, 'lib', 'ini_lint.py'), 'fixture.ini'], tmp)
        out['ini_lint report'] = hnorm(so)
        for fn in sorted(os.listdir(xml_dir)) if os.path.isdir(xml_dir) else []:
            run([os.path.join(SKILL, 'lib', 'preview.py'), os.path.join('out', fn), 'pv'], tmp)
        pv = os.path.join(tmp, 'pv')
        for fn in sorted(os.listdir(pv)) if os.path.isdir(pv) else []:
            with open(os.path.join(pv, fn), encoding='utf-8') as f:
                out[f'svg {fn}'] = hnorm(f.read())
        # positive control: a broken link MUST fail validation
        src = os.path.join(xml_dir, 'BPA Process - Example - Claim Loop.xml')
        neg = os.path.join(tmp, 'neg')
        os.makedirs(neg)
        with open(src, encoding='utf-8') as f:
            x = f.read()
        x = re.sub(r'(<onsuccess>)[0-9a-f-]{36}(</onsuccess>)', r'\1deadbeef\2', x, count=1)
        with open(os.path.join(neg, 'broken.xml'), 'w', encoding='utf-8') as f:
            f.write(x)
        code, so, se = run([os.path.join(SKILL, 'lib', 'validate.py'), 'neg'], tmp)
        out['positive control (broken link must fail)'] = 'fails as it must' if code == 1 else 'NOT CAUGHT'
    finally:
        shutil.rmtree(tmp, ignore_errors=True)
    return out


def main():
    argv = sys.argv[1:]
    if '--record' in argv:
        print(json.dumps(behaviour(), sort_keys=True))
        return 0
    ok = True
    print(f'skill-rpa-bp selftest  (python {sys.version.split()[0]})')
    bad = check_files()
    print(f'[{"FAIL" if bad else "PASS"}] files: {len(MANIFEST) - len(bad)}/{len(MANIFEST)} identical')
    for b in bad:
        print('       ' + b)
    ok &= not bad
    got = behaviour()
    diffs = [k for k in sorted(GOLDEN) if got.get(k) != GOLDEN[k]]
    print(f'[{"FAIL" if diffs else "PASS"}] behaviour: {len(GOLDEN) - len(diffs)}/{len(GOLDEN)} outputs match')
    for k in diffs:
        g = got.get(k, '<missing>')
        print(f'       DIFF {k}: {g if len(g) < 60 else g[:12] + "..."}')
    ok &= not diffs
    passthrough = [a for a in argv if a != '--record']
    code, so, se = run([os.path.join(SKILL, 'lib', 'compile_check.py'), '--self-test'] + passthrough, SKILL)
    langs = [l.strip() for l in so.splitlines() if l.strip().startswith(('csharp', 'visualbasic'))]
    if 'SELF-TEST PASS' in so:
        print('[PASS] compile gate: ' + '; '.join(langs))
    elif all('NO COMPILER' in l for l in langs) and langs:
        print('[SKIP] compile gate: no C#/VB compiler on this machine')
    else:
        print('[FAIL] compile gate: ' + ('; '.join(langs) or (so + se).strip()[-150:]))
        ok = False
    print('SELFTEST PASS' if ok else 'SELFTEST FAIL')
    return 0 if ok else 1


if __name__ == '__main__':
    sys.exit(main())
